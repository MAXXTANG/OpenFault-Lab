cmake_minimum_required(VERSION 3.20)
project(openfault_host C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(openfault firmware/common/openfault.c firmware/common/telemetry.c)
target_include_directories(openfault PUBLIC firmware/common)
target_compile_options(openfault PRIVATE -Wall -Wextra -Werror -Wconversion)
if(NOT MSVC)
  target_link_libraries(openfault PUBLIC m)
endif()
include(CTest)
add_executable(test_core tests/test_core.c)
target_link_libraries(test_core PRIVATE openfault)
target_compile_options(test_core PRIVATE -Wall -Wextra -Werror)
add_test(NAME core COMMAND test_core)
add_executable(synthetic_demo tools/synthetic_demo.c)
target_link_libraries(synthetic_demo PRIVATE openfault)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME synthetic_json COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/check_demo.py" $<TARGET_FILE:synthetic_demo>)
